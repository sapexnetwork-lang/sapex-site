$old = 'No tracking or advertising cookies are used\.'
$new = 'We also use advertising cookies to show relevant ads. You can manage your preferences below.'

Get-ChildItem -Path . -Filter *.html -Recurse |
    Where-Object { $_.Name -ne 'app.html' } |
    ForEach-Object {
        $content = Get-Content $_.FullName -Raw
        if ($content -match $old) {
            ($content -replace $old, $new) | Set-Content -Path $_.FullName -NoNewline
            Write-Host "Updated: $($_.Name)"
        }
    }
